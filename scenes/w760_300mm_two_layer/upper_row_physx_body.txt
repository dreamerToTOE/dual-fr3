        def close_upper_side_seal(side, phase="upper_B_seal"):
            """Remove residual pusher preload before enabling the native side suction joint."""
            from fr3_surface_gripper import ATTACHMENT_PARAMETERS
            from omni.isaac.dynamic_control import _dynamic_control
            from omni.physx import get_physx_scene_query_interface

            q = joint_state()[side]
            position, rotation = tcp_fk(side, q)
            sign = -1 if side == "left" else 1
            inward = sign * rotation[:, 1]
            if abs(inward[2]) > .05:
                raise RuntimeError(f"{side} seal axis differs from the horizontal side face")
            retreat = 0.
            stable = 0
            for _ in range(360):
                q = cartesian_joint_target(side, q, position - retreat * inward, rotation)
                command_positions(side, q)
                step()
                check_current_collision(phase + "_unload")
                wrench = np.asarray(last_force_sample[side]["tcp_gravity_corrected_world"])
                compression = -float(wrench[:3] @ inward)
                if np.linalg.norm(wrench[:3]) > 9. or np.linalg.norm(wrench[3:]) > 1.:
                    raise RuntimeError(f"{side} side seal unloading overload: {wrench.tolist()}")
                if compression > .12 or retreat < .00040:
                    retreat += .000005
                    if retreat > .00075:
                        raise RuntimeError(f"{side} side seal could not unload within 0.75 mm")
                quiet = np.max(np.abs(last_force_sample[side]["qdot_rad_s"])) < .006
                stable = stable + 1 if retreat >= .00040 and abs(compression) <= .12 and quiet else 0
                if stable == 16:
                    break
            else:
                raise RuntimeError(f"{side} side seal did not reach stationary low preload")

            # Replicate the native ray so the recorded attachment can be audited before stepping.
            dc = _dynamic_control.acquire_dynamic_control_interface()
            parent = f"/World/{side}_fr3/ABSideTool"
            pose = dc.get_rigid_body_pose(dc.get_rigid_body(parent))
            physical_rotation = Rotation.from_quat(list(pose.r)).as_matrix()
            offset = force_recorder.tcp_offsets[side].copy()
            offset[1] -= sign * ATTACHMENT_PARAMETERS["ray_standoff_m"]
            origin = np.asarray(list(pose.p)) + physical_rotation @ offset
            direction = sign * physical_rotation[:, 1]
            ray = get_physx_scene_query_interface().raycast_closest(
                tuple(origin), tuple(direction), ATTACHMENT_PARAMETERS["ray_range_m"])
            if not ray["hit"] or str(ray["rigidBody"]) != active_cube_path:
                raise RuntimeError(f"{side} stationary side-suction ray missed {active_cube_path}: {ray}")
            if not grippers.close(side):
                raise RuntimeError(f"{side} native side seal did not close")
            record(phase, "CLOSED_AT_LOW_PRELOAD", arm=side, body=str(ray["rigidBody"]),
                   ray_mm=1000 * ray["distance"], retraction_mm=retreat*1000,
                   normal_preload_N=compression)
            for _ in range(12):
                step()
                for arm in robots:
                    wrench = np.asarray(last_force_sample[arm]["tcp_gravity_corrected_world"])
                    if np.linalg.norm(wrench[:3]) > 9. or np.linalg.norm(wrench[3:]) > 1.:
                        raise RuntimeError(f"{arm} stationary side seal overload: {wrench.tolist()}")
                if not grippers.is_closed(side):
                    raise RuntimeError(f"{side} seal broke while holding the docking pose")
                check_current_collision(phase + "_hold")

        def finish_upper_retreat(after_release):
            weight = CONFIG["physics"]["cube_mass_kg"] * abs(CONFIG["physics"]["gravity_m_s2"][2])
            supported_load = after_release["median_contact_impulse_Ns"] * 120
            load_fraction = supported_load / weight
            report["upper_row"]["latest_release_support"] = {
                **after_release, "median_support_load_N": supported_load, "support_load_fraction": load_fraction,
            }
            if after_release["positive_steps"] < 48 or not .65 <= load_fraction <= 1.35:
                raise RuntimeError("Upper A release did not settle on the lower row "
                                   f"({after_release['positive_steps']}/60 support ticks, "
                                   f"median support {supported_load:.3f} N of {weight:.3f} N)")
            for phase in GUIDE["phases"][FIRST_LOADED:]:
                if not phase["carrying"]:
                    follow_side_guide(phase)
            allowed_contacts.clear()
            park_after_side_transfer()

        def slot3_cup_edge_margin(tcp, tool_rotation, center, box_rotation):
            # All four circular lips lie in the tool X-Z plane at the side-face TCP.
            span = ATTACHMENT_PARAMETERS["cup_center_half_span_m"]
            radius = ATTACHMENT_PARAMETERS["cup_radius_m"]
            basis = np.asarray(box_rotation).T @ np.asarray(tool_rotation)[:, [0, 2]]
            face_tcp = np.asarray(box_rotation).T @ (np.asarray(tcp) - np.asarray(center))
            face_extent = span * np.abs(basis).sum(axis=1) + radius * np.linalg.norm(basis, axis=1)
            return float(np.min(half - np.abs(face_tcp[1:]) - face_extent[1:]))

        def center_slot_edges(upper_seen):
            tracked = {name: (position, rotation) for name, position, rotation in upper_seen}
            left, left_rotation = tracked["Cube_9"]
            right, right_rotation = tracked["Cube_10"]
            return (float(left[0] + half * np.abs(left_rotation[0]).sum()),
                    float(right[0] - half * np.abs(right_rotation[0]).sum()))

        def fixed_x_entry_clearance(upper_seen, center, rotation):
            left_edge, right_edge = center_slot_edges(upper_seen)
            half_width = half * float(np.abs(rotation[0]).sum())
            x = float(center[0])
            left_gap = x - half_width - left_edge
            right_gap = right_edge - x - half_width
            # The existing 2 mm total allowance is shared by the two sides of the straight path.
            enough = bool(np.all(np.isfinite([left_gap, right_gap])) and min(left_gap, right_gap) >= .001 - 1e-12)
            return enough, {"left_gap_m": left_gap, "right_gap_m": right_gap,
                            "projected_cube_width_m": 2 * half_width, "slot_width_m": right_edge - left_edge,
                            "left_inner_edge_x_m": left_edge, "right_inner_edge_x_m": right_edge,
                            "observed_cube_x_m": x, "minimum_gap_per_side_m": .001,
                            "observed_yaw_deg": math.degrees(math.atan2(rotation[1, 0], rotation[0, 0]))}

        def find_upper_target(slot, upper_seen):
            x = (-.305, .305, -.155, .155, 0.)[slot - 1]
            if slot in (3, 4):
                tracked = {name: (position, rotation) for name, position, rotation in upper_seen}
                pos, rotation = tracked["Cube_7" if slot == 3 else "Cube_8"]
                extent = half * np.abs(rotation[0]).sum()
                x = pos[0] + (1 if slot == 3 else -1) * (extent + half)
            elif slot == 5:
                left_edge, right_edge = center_slot_edges(upper_seen)
                if right_edge - left_edge < 2 * half + .002 - 1e-12:
                    raise RuntimeError("Upper slot 5 camera measurement has less than 2 mm insertion clearance")
                x = (left_edge + right_edge) / 2
            return np.array([x, 1.076, SUPPORT_TOP + half])

        slot3_guide = None
        if 3 in RUNTIME_GUIDES and START_SLOT <= 3 <= CHECKPOINT.get("fixture_last_slot", 5):
